"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import * as THREE from "three";
import { createNoise3D } from "simplex-noise";

function fractionate(val: number, minVal: number, maxVal: number) {
  return (val - minVal) / (maxVal - minVal);
}

function modulate(
  val: number,
  minVal: number,
  maxVal: number,
  outMin: number,
  outMax: number,
) {
  const fr = fractionate(val, minVal, maxVal);
  const delta = outMax - outMin;
  return outMin + fr * delta;
}

function avg(arr: Uint8Array) {
  if (arr.length === 0) return 0;
  let total = 0;
  for (let i = 0; i < arr.length; i += 1) total += arr[i];
  return total / arr.length;
}

function max(arr: Uint8Array) {
  if (arr.length === 0) return 0;
  let current = arr[0];
  for (let i = 1; i < arr.length; i += 1) {
    if (arr[i] > current) current = arr[i];
  }
  return current;
}

export default function Page() {
  const visualizerRef = useRef<HTMLDivElement | null>(null);
  const audioRef = useRef<HTMLAudioElement | null>(null);

  const audioContextRef = useRef<AudioContext | null>(null);
  const audioAnalyserRef = useRef<AnalyserNode | null>(null);
  const audioSourceRef = useRef<MediaElementAudioSourceNode | null>(null);
  const micAnalyserRef = useRef<AnalyserNode | null>(null);
  const micSourceRef = useRef<MediaStreamAudioSourceNode | null>(null);
  const dataArrayRef = useRef<Uint8Array>(new Uint8Array(0));
  const micDataArrayRef = useRef<Uint8Array>(new Uint8Array(0));
  const audioUrlRef = useRef<string>("");
  const micStreamRef = useRef<MediaStream | null>(null);

  const rendererRef = useRef<THREE.WebGLRenderer | null>(null);
  const cameraRef = useRef<THREE.PerspectiveCamera | null>(null);
  const sceneRef = useRef<THREE.Scene | null>(null);
  const sphereRef = useRef<THREE.Mesh | null>(null);
  const baseDirRef = useRef<Float32Array | null>(null);
  const particleRef = useRef<THREE.Points | null>(null);
  const particleBaseRef = useRef<Float32Array | null>(null);
  const particleMaskRef = useRef<Float32Array | null>(null);
  const rafRef = useRef<number | null>(null);

  const [audioUrl, setAudioUrl] = useState("");
  const [isPlaying, setIsPlaying] = useState(false);
  const [hasFile, setHasFile] = useState(false);
  const [isMicLive, setIsMicLive] = useState(false);

  const noise3D = useRef(createNoise3D());

  const ensureAudioGraph = useCallback(async () => {
    const audio = audioRef.current;
    if (!audio) return null;

    if (!audioContextRef.current) {
      audioContextRef.current = new AudioContext();
    }

    const context = audioContextRef.current;
    if (context.state !== "running") {
      try {
        await context.resume();
      } catch {
        return null;
      }
    }

    if (!audioAnalyserRef.current) {
      const analyser = context.createAnalyser();
      analyser.fftSize = 512;
      audioAnalyserRef.current = analyser;
      dataArrayRef.current = new Uint8Array(analyser.frequencyBinCount);
    }

    if (!audioSourceRef.current) {
      try {
        audioSourceRef.current = context.createMediaElementSource(audio);
      } catch {
        return audioAnalyserRef.current;
      }
    }

    try {
      audioSourceRef.current.disconnect();
    } catch {
      // Ignore stale graph disconnect errors.
    }
    try {
      audioAnalyserRef.current.disconnect();
    } catch {
      // Ignore stale graph disconnect errors.
    }

    audioSourceRef.current.connect(audioAnalyserRef.current);
    audioAnalyserRef.current.connect(context.destination);

    return audioAnalyserRef.current;
  }, []);

  const ensureMicGraph = useCallback(async () => {
    if (!audioContextRef.current) {
      audioContextRef.current = new AudioContext();
    }

    const context = audioContextRef.current;
    if (context.state !== "running") {
      try {
        await context.resume();
      } catch {
        return null;
      }
    }

    if (!micAnalyserRef.current) {
      const analyser = context.createAnalyser();
      analyser.fftSize = 512;
      micAnalyserRef.current = analyser;
      micDataArrayRef.current = new Uint8Array(analyser.frequencyBinCount);
    }

    if (!micStreamRef.current) {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        micStreamRef.current = stream;
      } catch {
        return null;
      }
    }

    if (!micSourceRef.current) {
      try {
        micSourceRef.current = context.createMediaStreamSource(micStreamRef.current);
      } catch {
        return micAnalyserRef.current;
      }
    }

    try {
      micSourceRef.current.disconnect();
    } catch {
      // Ignore stale graph disconnect errors.
    }
    try {
      micAnalyserRef.current.disconnect();
    } catch {
      // Ignore stale graph disconnect errors.
    }

    micSourceRef.current.connect(micAnalyserRef.current);
    micAnalyserRef.current.connect(context.destination);
    setIsMicLive(true);

    return micAnalyserRef.current;
  }, []);

  const warpSphere = useCallback((bassFr: number, treFr: number, time: number) => {
    const sphere = sphereRef.current;
    if (!sphere) return;

    const geometry = sphere.geometry as THREE.IcosahedronGeometry;
    const positionAttr = geometry.getAttribute("position") as THREE.BufferAttribute;
    const baseDir = baseDirRef.current;
    if (!baseDir) return;

    const radius = 10;
    const amp = 5;
    const rf = 0.00005;

    for (let i = 0; i < positionAttr.count; i += 1) {
      const i3 = i * 3;
      const nx = baseDir[i3];
      const ny = baseDir[i3 + 1];
      const nz = baseDir[i3 + 2];

      const distance =
        radius +
        bassFr +
        noise3D.current(nx + time * rf * 4, ny + time * rf * 6, nz + time * rf * 7) *
          amp *
          treFr *
          2;

      positionAttr.setXYZ(i, nx * distance, ny * distance, nz * distance);
    }

    positionAttr.needsUpdate = true;
    geometry.computeVertexNormals();
  }, []);

  useEffect(() => {
    const mount = visualizerRef.current;
    if (!mount) return;

    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(75, 1, 0.1, 1000);
    camera.position.set(0, 0, 100);
    camera.lookAt(0, 0, 0);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 100));
    renderer.setClearColor("#000000");
    renderer.domElement.style.display = "block";
    renderer.domElement.style.margin = "0 auto";

    const geometry = new THREE.IcosahedronGeometry(30,8);
const material = new THREE.MeshStandardMaterial({
  color: "#ffffff", // darker base → glow stands out
  roughness: 0.2,
  metalness: 0.7,
  emissive: new THREE.Color("#ffffff"),
  emissiveIntensity: 0.05, // start at 0,
  wireframe:true
});

    const sphere = new THREE.Mesh(geometry, material);
    const light = new THREE.DirectionalLight("#ffffff", 2);
    light.position.set(0, 50, 100);
    const particleCount = 1200;
const positions = new Float32Array(particleCount * 3);
const particleBase = new Float32Array(particleCount * 3);
const particleMask = new Float32Array(particleCount);

for (let i = 0; i < particleCount; i++) {
  const r = 31 + Math.random() * 5;
  const theta = Math.random() * Math.PI * 2;
  const phi = Math.acos(Math.random() * 2 - 1);
  const mask = Math.random() < 0.16 ? 0.55 + Math.random() * 0.9 : 0;

  particleMask[i] = mask;

  const x = r * Math.sin(phi) * Math.cos(theta);
  const y = r * Math.sin(phi) * Math.sin(theta);
  const z = r * Math.cos(phi);

  positions[i * 3] = x;
  positions[i * 3 + 1] = y;
  positions[i * 3 + 2] = z;

  particleBase[i * 3] = x;
  particleBase[i * 3 + 1] = y;
  particleBase[i * 3 + 2] = z;
}

const particleGeo = new THREE.BufferGeometry();
particleGeo.setAttribute("position", new THREE.BufferAttribute(positions, 3));

const particleMat = new THREE.PointsMaterial({
  color: "#ff8800",
  size: 0.5,
  transparent: true,
  blending: THREE.AdditiveBlending,
  depthWrite: false,
});

const particles = new THREE.Points(particleGeo, particleMat);
scene.add(particles);

    scene.add(camera);
    scene.add(light);
    scene.add(sphere);

    const positionAttr = geometry.getAttribute("position") as THREE.BufferAttribute;
    const baseDir = new Float32Array(positionAttr.array.length);
    for (let i = 0; i < positionAttr.count; i += 1) {
      const i3 = i * 3;
      const x = positionAttr.getX(i);
      const y = positionAttr.getY(i);
      const z = positionAttr.getZ(i);
      const len = Math.sqrt(x * x + y * y + z * z) || 1;
      baseDir[i3] = x / len;
      baseDir[i3 + 1] = y / len;
      baseDir[i3 + 2] = z / len;
    }

    mount.appendChild(renderer.domElement);

    sceneRef.current = scene;
    cameraRef.current = camera;
    rendererRef.current = renderer;
    sphereRef.current = sphere;
    baseDirRef.current = baseDir;
    particleRef.current = particles;
    particleBaseRef.current = particleBase;
    particleMaskRef.current = particleMask;

    const resize = () => {
      const width = mount.clientWidth || window.innerWidth;
      const height = mount.clientHeight || window.innerHeight;
      renderer.setSize(width, height, false);
      camera.aspect = width / height;
      camera.updateProjectionMatrix();
    };

    resize();
    window.addEventListener("resize", resize);

    const render = () => {
      const audioAnalyser = audioAnalyserRef.current;
      const micAnalyser = micAnalyserRef.current;
      const dataArray = dataArrayRef.current;
      const micDataArray = micDataArrayRef.current;

      let bassFr = 0;
      let treFr = 0;
      let micBassFr = 0;
      let micTreFr = 0;

      if (audioAnalyser && dataArray.length > 0) {
        audioAnalyser.getByteFrequencyData(dataArray as any);

        const lowerHalf = dataArray.slice(0, Math.max(1, Math.floor(dataArray.length / 2) - 1));
        const upperHalf = dataArray.slice(
          Math.max(0, Math.floor(dataArray.length / 2) - 1),
          dataArray.length - 1,
        );

        const lowerMaxFr = max(lowerHalf) / Math.max(1, lowerHalf.length);
        const upperAvgFr = avg(upperHalf) / Math.max(1, upperHalf.length);

        bassFr = modulate(Math.pow(lowerMaxFr, 0.8), 0, 1, 0, 8);
        treFr = modulate(upperAvgFr, 0, 1, 0, 4);
      }

      if (micAnalyser && micDataArray.length > 0) {
        micAnalyser.getByteFrequencyData(micDataArray as any);

        const lowerHalf = micDataArray.slice(0, Math.max(1, Math.floor(micDataArray.length / 2) - 1));
        const upperHalf = micDataArray.slice(
          Math.max(0, Math.floor(micDataArray.length / 2) - 1),
          micDataArray.length - 1,
        );

        const lowerMaxFr = max(lowerHalf) / Math.max(1, lowerHalf.length);
        const upperAvgFr = avg(upperHalf) / Math.max(1, upperHalf.length);

        micBassFr = modulate(Math.pow(lowerMaxFr, 0.8), 0, 1, 0, 8);
        micTreFr = modulate(upperAvgFr, 0, 1, 0, 4);
      }

      const activeBassFr = Math.max(bassFr, micBassFr);
      const activeTreFr = Math.max(treFr, micTreFr);

      sphere.rotation.z+=0.005
      particles.rotation.y-=0.01
      const particleGeoAttr = particles.geometry.getAttribute("position") as THREE.BufferAttribute;
      const particleBase = particleBaseRef.current;
      const particleMask = particleMaskRef.current;
      if (particleBase && particleMask) {
        const breath = 1 + activeBassFr * 0.012 + activeTreFr * 0.006;
        for (let i = 0; i < particleGeoAttr.count; i += 1) {
          const i3 = i * 3;
          const x = particleBase[i3];
          const y = particleBase[i3 + 1];
          const z = particleBase[i3 + 2];
          const mask = particleMask[i];
          const wobble = mask > 0 ? noise3D.current(x * 0.04 + performance.now() * 0.00002, y * 0.04, z * 0.04) : 0;
          const radiusBoost = mask > 0 ? mask * (activeBassFr * 0.08 + activeTreFr * 0.04 + wobble * 0.45) : 0;
          const scale = breath + radiusBoost;
          particleGeoAttr.setXYZ(i, x * scale, y * scale, z * scale);
        }
        particleGeoAttr.needsUpdate = true;
      }
      const energy = activeBassFr * 0.6 + activeTreFr * 0.4;

// smooth it (important)
const glow = THREE.MathUtils.lerp(
  material.emissiveIntensity,
  energy * 1.5,
  0.1
);

const pulse = Math.sin(performance.now() * 0.003) * 0.3;

material.emissiveIntensity =
  glow + pulse * (0.2 + activeBassFr * 0.3);

      warpSphere(activeBassFr, activeTreFr, performance.now());
      renderer.render(scene, camera);
      rafRef.current = requestAnimationFrame(render);
      
    };

    rafRef.current = requestAnimationFrame(render);

    return () => {
      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
      window.removeEventListener("resize", resize);

      sphere.geometry.dispose();
      material.dispose();
      particles.geometry.dispose();
      particleMat.dispose();
      renderer.dispose();

      if (renderer.domElement.parentElement === mount) {
        mount.removeChild(renderer.domElement);
      }
    };
  }, [warpSphere]);

  const handleFileChange = useCallback(async (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (!file) return;

    if (!file.name.toLowerCase().endsWith(".mp3")) {
      alert("Please upload an mp3 file");
      return;
    }

    const nextUrl = URL.createObjectURL(file);
    setAudioUrl((prev) => {
      if (prev) URL.revokeObjectURL(prev);
      return nextUrl;
    });
    audioUrlRef.current = nextUrl;

    setHasFile(true);

    const audio = audioRef.current;
    if (!audio) return;

    audio.pause();
    audio.src = nextUrl;
    audio.load();

    await ensureAudioGraph();

    try {
      await audio.play();
      setIsPlaying(true);
    } catch {
      setIsPlaying(false);
    }
  }, [ensureAudioGraph]);

  const handleVisualizerClick = useCallback(async () => {
    if (!hasFile && !isMicLive) return;

    const audio = audioRef.current;
    if (hasFile && audio) {
      await ensureAudioGraph();

      if (audio.paused) {
        try {
          await audio.play();
          setIsPlaying(true);
        } catch {
          setIsPlaying(false);
        }
        return;
      }

      audio.pause();
      setIsPlaying(false);
    }

    if (!hasFile) {
      await ensureMicGraph();
    }
  }, [ensureAudioGraph, ensureMicGraph, hasFile, isMicLive]);

  const handleMicToggle = useCallback(async () => {
    if (isMicLive) {
      if (micSourceRef.current) {
        try {
          micSourceRef.current.disconnect();
        } catch {
          // Ignore disconnect races.
        }
        micSourceRef.current = null;
      }
      if (micAnalyserRef.current) {
        try {
          micAnalyserRef.current.disconnect();
        } catch {
          // Ignore disconnect races.
        }
      }
      if (micStreamRef.current) {
        micStreamRef.current.getTracks().forEach((track) => track.stop());
        micStreamRef.current = null;
      }
      micAnalyserRef.current = null;
      micDataArrayRef.current = new Uint8Array(0);
      setIsMicLive(false);
      return;
    }

    await ensureMicGraph();
  }, [ensureMicGraph, isMicLive]);

  useEffect(() => {
    const audio = audioRef.current;
    if (!audio) return;

    const onPlay = () => setIsPlaying(true);
    const onPause = () => setIsPlaying(false);
    const onEnded = () => setIsPlaying(false);

    audio.addEventListener("play", onPlay);
    audio.addEventListener("pause", onPause);
    audio.addEventListener("ended", onEnded);

    return () => {
      audio.removeEventListener("play", onPlay);
      audio.removeEventListener("pause", onPause);
      audio.removeEventListener("ended", onEnded);
    };
  }, []);

  useEffect(() => {
    return () => {
      if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);

      if (audioSourceRef.current) {
        try {
          audioSourceRef.current.disconnect();
        } catch {
          // Ignore stale graph disconnect errors.
        }
      }

      if (audioAnalyserRef.current) {
        try {
          audioAnalyserRef.current.disconnect();
        } catch {
          // Ignore stale graph disconnect errors.
        }
      }

      if (micSourceRef.current) {
        try {
          micSourceRef.current.disconnect();
        } catch {
          // Ignore stale graph disconnect errors.
        }
      }

      if (micAnalyserRef.current) {
        try {
          micAnalyserRef.current.disconnect();
        } catch {
          // Ignore stale graph disconnect errors.
        }
      }

      if (micStreamRef.current) {
        micStreamRef.current.getTracks().forEach((track) => track.stop());
        micStreamRef.current = null;
      }

      if (audioContextRef.current) {
        void audioContextRef.current.close();
      }
    };
  }, []);

  return (
    <main className="min-h-screen bg-black text-white">
      <div className="mx-auto flex min-h-screen w-full max-w-6xl flex-col gap-4 p-4 md:p-8">
        <div className="flex flex-wrap items-center gap-3">
          <label className="rounded-md border border-white/30 px-3 py-2 text-sm">
            Select MP3
            <input
              type="file"
              accept=".mp3"
              onChange={handleFileChange}
              className="ml-3 text-xs"
            />
          </label>
          <span className="text-sm text-white/70">Click the orb to play or pause</span>
          <span className="text-xs text-white/50">
            {hasFile ? (isPlaying ? "Playing" : "Paused") : "No file selected"}
          </span>
          <button
            type="button"
            onClick={handleMicToggle}
            className="rounded-md border border-white/30 px-3 py-2 text-sm"
          >
            {isMicLive ? "Stop Mic" : "Start Mic"}
          </button>
        </div>

        <button
          type="button"
          onClick={handleVisualizerClick}
          className="relative h-[70vh] min-h-[420px] w-full overflow-hidden rounded-xl border border-white/15"
          aria-label="Audio visualizer orb"
        >
          <div ref={visualizerRef} className="h-full w-full flex items-center justify-center" />
          {!isPlaying && (
            <div className="pointer-events-none absolute inset-0 flex items-center justify-center text-lg text-white/80">
              {hasFile ? "Click to play" : "Select an MP3 file"}
            </div>
          )}
        </button>

        <audio ref={audioRef} controls className="w-full" preload="auto" />
      </div>
    </main>
  );
}
